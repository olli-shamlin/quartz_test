1. git clone https://github.com/jackyzha0/quartz.git
2. cd quartz
3. npm i
4. npx quartz create
	1. "Choose how to initialize ...": Empty Quartz
	2. "Choose how Quartz should resolve links ...": Treat links as shortest path
5. git remote -v
6. git remote remove origin
7. git remote -v
8. git remote add origin [YOUR-REPO]
9. git remove -v
10. npx quartz sync --no-pull
